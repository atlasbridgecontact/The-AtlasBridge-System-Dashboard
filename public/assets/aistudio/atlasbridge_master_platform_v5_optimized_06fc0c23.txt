        function evaluateEligibility() {
            syncWizardInput();
            eligiblePathIds = [];

            const isHighIncome = (clientData.filingStatus === 'married' ? clientData.grossAGI > 200000 : clientData.grossAGI > 150000) || clientData.fplPct > 500;
            const isOutsideWindow = isOutsideAcaEnrollmentWindow();

            const outsideNotice = document.getElementById('outside-aca-window-notice');
            if (isOutsideWindow && selectedCategory === 'health') {
                if (outsideNotice) outsideNotice.classList.remove('hidden');
            } else {
                if (outsideNotice) outsideNotice.classList.add('hidden');
            }

            const highIncomeNotice = document.getElementById('high-income-notice');
            if (isHighIncome && selectedCategory === 'health' && !isOutsideWindow) {
                if (highIncomeNotice) {
                    highIncomeNotice.classList.remove('hidden');
                    document.getElementById('high-income-val').textContent = '$' + clientData.grossAGI.toLocaleString() + ' (' + clientData.fplPct + '% FPL)';
                }
            } else {
                if (highIncomeNotice) highIncomeNotice.classList.add('hidden');
            }

            if (selectedCategory === 'unwanted_life') {
                eligiblePathIds.push('unwanted_life', 'annuities', 'rollovers');
            } else if (selectedCategory === 'health') {
                if (isOutsideWindow) {
                    eligiblePathIds.push('private_health', 'traditional_aca');
                } else {
                    if (!isHighIncome && clientData.employerGroup !== 'yes_affordable' && !clientData.active401k && clientData.fplPct <= 500) {
                        eligiblePathIds.push('atlasbridge');
                    }
                    if (clientData.employerGroup !== 'yes_affordable') eligiblePathIds.push('traditional_aca');
                    if (clientData.healthStatus === 'healthy' || clientData.healthStatus === 'moderate' || isHighIncome) {
                        eligiblePathIds.push('private_health');
                    }
                }
                if (clientData.ownsBusiness || clientData.w2Employees >= 1) eligiblePathIds.push('group');
            } else if (selectedCategory === 'life') {
                eligiblePathIds.push('life');
                if (clientData.lifePolicyStatus !== 'no') eligiblePathIds.push('unwanted_life');
            } else if (selectedCategory === 'annuities') {
                eligiblePathIds.push('annuities');
                if (clientData.lifePolicyStatus !== 'no') eligiblePathIds.push('unwanted_life');
            } else if (selectedCategory === 'rollovers') {
                eligiblePathIds.push('rollovers');
                if (clientData.lifePolicyStatus !== 'no') eligiblePathIds.push('unwanted_life');
            } else if (selectedCategory === 'settlements') {
                if (clientData.lifePolicyStatus !== 'no') {
                    eligiblePathIds.push('settlements', 'unwanted_life');
                }
            }

            if (clientData.lifePolicyStatus === 'no' && selectedCategory !== 'unwanted_life') {
                eligiblePathIds = eligiblePathIds.filter(id => id !== 'unwanted_life' && id !== 'settlements');
            }

            if (eligiblePathIds.length === 0) {
                if (selectedCategory === 'unwanted_life') eligiblePathIds.push('unwanted_life');
                else if (selectedCategory === 'life') eligiblePathIds.push('life');
                else if (isOutsideWindow) eligiblePathIds.push('private_health');
                else if (isHighIncome) eligiblePathIds.push('private_health', 'traditional_aca');
                else eligiblePathIds.push('atlasbridge');
            }

            wizardCompleted = true;
            SoundFX.playUnlock();

            document.getElementById('header-tax-year-display').textContent = clientData.taxYear + ' Open Enrollment';
            document.getElementById('header-tax-year-display').className = 'text-white font-extrabold';

            document.getElementById('wizard-section').classList.add('hidden');
            document.getElementById('unlocked-dashboards-section').classList.remove('hidden');

            if (eligiblePathIds.length > 0) {
                currentActiveDashboardId = eligiblePathIds[0];
            }

            updateUI();
            loadPathDashboard(currentActiveDashboardId);

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function populatePolicyTypeDropdownForTab(tabKey) {
            const dropdown = document.getElementById('proj-policy-type');
            const creditingWrapper = document.getElementById('crediting-strategy-wrapper');
            const timeHorizonWrapper = document.getElementById('time-horizon-wrapper');
            if (!dropdown) return;
            dropdown.innerHTML = '';

            let options = [];
            if (tabKey === 'term') {
                options = [
                    { val: 'term_10', label: '10-Year Level Term' },
                    { val: 'term_20', label: '20-Year Level Term (Recommended)' },
                    { val: 'term_30', label: '30-Year Level Term' }
                ];
                if (creditingWrapper) creditingWrapper.style.display = 'none';
                if (timeHorizonWrapper) timeHorizonWrapper.style.display = 'none';
            } else if (tabKey === 'whole') {
                options = [
                    { val: 'whole_standard', label: 'Standard Participating Whole Life (Dividends)' },
                    { val: 'whole_paid_up', label: 'Single Premium / Paid-Up Additions (PUA)' }
                ];
                if (creditingWrapper) creditingWrapper.style.display = 'none';
                if (timeHorizonWrapper) timeHorizonWrapper.style.display = 'block';
            } else if (tabKey === 'universal') {
                options = [
                    { val: 'iul', label: 'Indexed Universal Life (IUL - S&P 500 Indexed)' },
                    { val: 'vul', label: 'Variable Universal Life (VUL - Subaccounts)' },
                    { val: 'gul', label: 'Guaranteed Universal Life (GUL - Age 120 Lock)' }
                ];
                if (creditingWrapper) creditingWrapper.style.display = 'block';
                if (timeHorizonWrapper) timeHorizonWrapper.style.display = 'block';
            } else if (tabKey === 'other') {
                options = [
                    { val: 'key_person', label: 'Key-Person Business Protection' },
                    { val: 'buy_sell', label: 'Buy-Sell Partnership Agreement' },
                    { val: 'exec_bonus', label: 'Executive Bonus (Sec. 162 Plan)' },
                    { val: 'final_expense', label: 'Senior Final Expense Whole Life' }
                ];
                if (creditingWrapper) creditingWrapper.style.display = 'none';
                if (timeHorizonWrapper) timeHorizonWrapper.style.display = 'block';
            }

            options.forEach(opt => {
                const el = document.createElement('option');
                el.value = opt.val;
                el.textContent = opt.label;
                dropdown.appendChild(el);

        function updateLifeInsuranceCalc() {
            const db = parseFloat(document.getElementById('proj-db-input')?.value) || 500000;
            const age = clientData.clientAge;

            let carrierFactor = 1.0;
            if (selectedLifeCarrier === 'Corebridge Financial') carrierFactor = 1.07;
            else if (selectedLifeCarrier === 'Lincoln Financial') carrierFactor = 1.12;
            else if (selectedLifeCarrier === 'Pacific Life') carrierFactor = 1.03;
            else if (selectedLifeCarrier === 'Principal') carrierFactor = 1.09;
            else carrierFactor = 1.0;

            let riderMultiplier = 1.0 + (riderSliderBudget / 100);
            const ageFactor = Math.pow(1.045, Math.max(0, age - 30));

            let baseRate = 4.2;
            if (activeLifeTab === 'whole') baseRate = 16.0;
            if (activeLifeTab === 'universal') baseRate = 24.0;
            if (activeLifeTab === 'other') {
                const subType = document.getElementById('proj-policy-type')?.value;
                if (subType === 'final_expense') baseRate = 35.0;
                else if (subType === 'exec_bonus') baseRate = 28.0;
                else baseRate = 12.0;
            }

            const basePremMonthly = Math.round((db / 100000) * baseRate * ageFactor * riderMultiplier * carrierFactor);

            let timeHorizon = 20;
            if (activeLifeTab === 'term') {
                const selectedTermType = document.getElementById('proj-policy-type')?.value;
                if (selectedTermType === 'term_10') timeHorizon = 10;
                else if (selectedTermType === 'term_30') timeHorizon = 30;
                else timeHorizon = 20;
            } else {
                timeHorizon = parseInt(document.getElementById('proj-time-period')?.value) || 20;
            }

            let growthRate = 0.068;
            if (selectedCreditingStrategy === 'sp500_monthly_8.2') growthRate = 0.078;
            else if (selectedCreditingStrategy === 'fixed_floor_4.5') growthRate = 0.045;
            else if (selectedCreditingStrategy === 'blended_dynamic') growthRate = 0.085;

            if (selectedLifeCarrier === 'Pacific Life') growthRate += 0.007;
            if (selectedLifeCarrier === 'Corebridge Financial') growthRate += 0.005;

            let estCashValue = 0;
            if (activeLifeTab !== 'term') {
                let currentCV = 0;
                const annualContribution = basePremMonthly * 12;
                for (let i = 1; i <= timeHorizon; i++) {
                    currentCV = (currentCV + annualContribution * 0.85) * (1 + growthRate);
                }
                estCashValue = Math.round(currentCV);
            } else {
                estCashValue = 0;
            }

            const maxLiving = Math.round(db * (selectedAccelerationTrigger / 100));

            const kpi1Val = document.getElementById('kpi-card-1-val');
            const kpi1Sub = document.getElementById('kpi-card-1-sub');
            const kpi2Val = document.getElementById('kpi-card-2-val');
            const kpi3Val = document.getElementById('kpi-card-3-val');
            const kpi4Val = document.getElementById('kpi-card-4-val');

            if (kpi1Sub) kpi1Sub.textContent = `${selectedLifeCarrier} (Active)`;

            if (activeLifeTab === 'term') {
                if (kpi1Val) kpi1Val.textContent = '$' + basePremMonthly.toLocaleString() + ' / mo';
                if (kpi2Val) kpi2Val.textContent = '$' + db.toLocaleString();
                if (kpi3Val) kpi3Val.textContent = '$' + maxLiving.toLocaleString();
                if (kpi4Val) kpi4Val.textContent = `${timeHorizon}-Yr Level Term`;
            } else if (activeLifeTab === 'universal') {
                if (kpi1Val) kpi1Val.textContent = '$' + basePremMonthly.toLocaleString() + ' / mo';
                if (kpi2Val) kpi2Val.textContent = '$' + estCashValue.toLocaleString();
                if (kpi3Val) kpi3Val.textContent = '0% Floor Protected';
                if (kpi4Val) kpi4Val.textContent = 'Tax-Free Loans';
            } else if (activeLifeTab === 'whole') {
                if (kpi1Val) kpi1Val.textContent = '$' + basePremMonthly.toLocaleString() + ' / mo';
                if (kpi2Val) kpi2Val.textContent = 'Guaranteed to Age 120';
                if (kpi3Val) kpi3Val.textContent = '$' + db.toLocaleString();
                if (kpi4Val) kpi4Val.textContent = '4.5% - 5.5% Div.';
            } else {
                if (kpi1Val) kpi1Val.textContent = '$' + basePremMonthly.toLocaleString() + ' / mo';
                if (kpi2Val) kpi2Val.textContent = '$' + db.toLocaleString();
                if (kpi3Val) kpi3Val.textContent = '$' + estCashValue.toLocaleString();
                if (kpi4Val) kpi4Val.textContent = `${timeHorizon}-Year Target`;
            }

            const projBoxCash = document.getElementById('proj-box-cash-display');
            if (projBoxCash) projBoxCash.textContent = '$' + estCashValue.toLocaleString();

            const summaryCashVal = document.getElementById('summary-cash-value-display');
            if (summaryCashVal) summaryCashVal.textContent = '$' + estCashValue.toLocaleString();

            updateLifeChartData(basePremMonthly, timeHorizon, growthRate);
        }

        function updateLifeChartData(monthlyPrem, maxYears = 20, growthRate = 0.068) {
            if (!lifeChart) return;
            const annualPrem = monthlyPrem * 12;
            const years = [1, 5, 10, 15, 20, 25, 30].filter(y => y <= maxYears || y === 1 || y === 5);

            const cumPremiums = years.map(y => Math.round(annualPrem * y));
            
            const cashGrowth = years.map(y => {
                if (activeLifeTab === 'term') {
                    return 0;
                }
                let cv = 0;
                for (let i = 1; i <= y; i++) {
                    cv = (cv + annualPrem * 0.85) * (1 + growthRate);
                }
                return Math.round(cv);
            });

            lifeChart.data.labels = years.map(y => `Yr ${y}`);
            lifeChart.data.datasets[0].data = cumPremiums;
            lifeChart.data.datasets[1].data = cashGrowth;
            lifeChart.data.datasets[1].label = activeLifeTab === 'term' ? 'Policy Cash Value ($0 for Term)' : `Policy Cash Value Growth (${selectedLifeCarrier})`;
            lifeChart.update();
        }

        function onAnnuityVarietyChange(varId) {
            const taxClass = document.getElementById('annuity-tax-class').value;
            const list = ANNUITY_CLASSIFICATIONS[taxClass];
            const spec = list.find(i => i.id === varId) || list[0];
            currentActiveAnnuitySpec = spec;

            document.getElementById('annuity-desc-lbl').textContent = spec.desc;

            const statusSelect = document.getElementById('annuity-status-option');
            if (statusSelect && spec.defaultStatus) {
                statusSelect.value = spec.defaultStatus;
            }

            let initialDep = spec.defaultDep;
            if (taxClass === 'qualified' && clientData.dormantRetirement > 0) {
                initialDep = Math.max(spec.minDep, clientData.dormantRetirement);
            } else if (taxClass === 'non_qualified' && (clientData.existingAnnuityCash > 0 || clientData.liquidSavings > 0)) {
                initialDep = Math.max(spec.minDep, clientData.existingAnnuityCash || clientData.liquidSavings);
            }

            const slider = document.getElementById('annuity-inv-slider');
            slider.min = spec.minDep; slider.max = spec.maxDep; slider.step = spec.stepDep; slider.value = initialDep;
            document.getElementById('annuity-deposit-manual').value = initialDep;

            const yearsSelect = document.getElementById('annuity-years-select');
            yearsSelect.innerHTML = '';
            spec.years.forEach(y => {
                const opt = document.createElement('option');
                opt.value = y;
                opt.textContent = y === 0 ? "0 Years (Immediate Income)" : `${y} Year Accumulation`;
                if (y === spec.defaultYears) opt.selected = true;
                yearsSelect.appendChild(opt);

            const carrierSelectVal = document.getElementById('annuity-carrier-interactive-select')?.value || spec.carrier || "Athene USA";
            onAnnuityCarrierInteractiveChange(carrierSelectVal);

            calculateAnnuityProjection();
        }

        function calculateAnnuityProjection() {
            if (!currentActiveAnnuitySpec) return;
            const spec = currentActiveAnnuitySpec;
            const inv = parseFloat(document.getElementById('annuity-deposit-manual').value) || spec.defaultDep;
            const years = parseInt(document.getElementById('annuity-years-select').value) || 0;
            
            const carrierSelected = document.getElementById('annuity-carrier-interactive-select')?.value || 'Athene USA';
            let carrierBoost = 1.0;
            if (carrierSelected.includes('Athene')) carrierBoost = 1.02;
            if (carrierSelected.includes('Allianz')) carrierBoost = 1.015;
            if (carrierSelected.includes('SILAC')) carrierBoost = 1.03;

            const rate = (spec.fixedRate / 100) * carrierBoost;
            const clientAge = clientData.clientAge;
            const targetAge = clientAge + years;

            const ageEl = document.getElementById('annuity-target-age-lbl');
            if (ageEl) ageEl.textContent = targetAge + ' Years Old';

            const yearsLbl = document.getElementById('annuity-years-lbl');
            if (yearsLbl) yearsLbl.textContent = years === 0 ? "Immediate Start" : `${years} Years`;

            let pav = inv;
            if (years > 0) pav = inv * Math.pow((1 + rate), years);

            const rawGains = Math.max(0, pav - inv);
            const estPayout = years === 0 ? (inv * 0.074 * carrierBoost) : (pav * 0.066 * carrierBoost);

            const pEl = document.getElementById('annuity-res-principal');
            if (pEl) pEl.textContent = '$' + inv.toLocaleString();
            const gEl = document.getElementById('annuity-res-raw-gains');
            if (gEl) gEl.textContent = '+$' + Math.round(rawGains).toLocaleString();
            const pavEl = document.getElementById('annuity-res-pav');
            if (pavEl) pavEl.textContent = '$' + Math.round(pav).toLocaleString();
            const payEl = document.getElementById('annuity-res-payout');
            if (payEl) payEl.textContent = '$' + Math.round(estPayout).toLocaleString() + ' / yr';
        }

        function getACAStats(fplPct, famMult) {
            if (fplPct <= 150) {
                return { monthlyACA: Math.round(25 * famMult), deductible: Math.round(500 * famMult), oopMax: Math.round(2000 * famMult), avText: "94% Silver CSR", tierName: "Silver CSR 94% Variant" };
            } else if (fplPct <= 200) {
                return { monthlyACA: Math.round(75 * famMult), deductible: Math.round(1000 * famMult), oopMax: Math.round(3500 * famMult), avText: "87% Silver CSR", tierName: "Silver CSR 87% Variant" };
            } else if (fplPct <= 250) {
                return { monthlyACA: Math.round(120 * famMult), deductible: Math.round(2500 * famMult), oopMax: Math.round(5800 * famMult), avText: "73% Enhanced", tierName: "Enhanced Silver CSR 73%" };
            } else if (fplPct <= 400) {
                const estPrem = Math.round(Math.max(140, (fplPct * 1.6)) * famMult);
                return { monthlyACA: estPrem, deductible: Math.round(4200 * famMult), oopMax: Math.round(7800 * famMult), avText: "70% Standard Silver", tierName: "Standard Silver Plan (70% AV)" };
            } else {
                return { monthlyACA: Math.round(550 * famMult), deductible: Math.round(8500 * famMult), oopMax: Math.round(9450 * famMult), avText: "Full Price Commercial", tierName: "Full-Price Commercial Plan (>400% FPL)" };
            }
        }

        function recalculateFullAtlasbridge() {
            const hhSize = clientData.householdSize;
            const fpl100 = get100PercentFPL(hhSize);
            const famMult = hhSize > 1 ? 1.8 : 1.0;
            const isMarried = clientData.filingStatus === 'married';

            const currentIRSLimit = getIRSLimitForAge(clientData.clientAge, clientData.filingStatus);

            const maxAllowedContrib = Math.min(currentIRSLimit, Math.max(0, currentAGI - fpl100));
            const contribSlider = document.getElementById('master-contrib-slider');
            const manualContribInput = document.getElementById('manual-contrib-input');

            if (contribSlider) {
                contribSlider.max = currentIRSLimit;
                if (targetAnnualContribution > maxAllowedContrib) {
                    targetAnnualContribution = maxAllowedContrib;
                    contribSlider.value = targetAnnualContribution;
                    if (manualContribInput) manualContribInput.value = targetAnnualContribution;
                }
            }
            if (manualContribInput) {
                manualContribInput.max = currentIRSLimit;
            }

            const noticeLbl = document.getElementById('contrib-policy-notice-lbl');
            if (noticeLbl) {
                noticeLbl.innerHTML = `<i class="fa-solid fa-circle-info mr-1"></i> Capped at $${currentIRSLimit.toLocaleString()} max IRS limit (${isMarried ? 'Married Joint Dual Cap' : (clientData.clientAge >= 50 ? 'Age 50+ Catch-up Cap' : 'Standard Cap <50')}).`;
            }

            const optimizedMAGI = Math.max(fpl100, currentAGI - targetAnnualContribution);
            const optimizedFPL = Math.round((optimizedMAGI / fpl100) * 100);

            const unoptFPL = Math.round((currentAGI / fpl100) * 100);
            const unoptStats = getACAStats(unoptFPL, famMult);
            const acaStats = getACAStats(optimizedFPL, famMult);

            const carrierObj = REGIONAL_HEALTH_CARRIERS.find(c => c.id === clientData.selectedHealthCarrier) || REGIONAL_HEALTH_CARRIERS[0];
            const displayCarrier = carrierObj.name;

            let displayPlanName = acaStats.tierName;
            let displayPrem = acaStats.monthlyACA;
            let displayDed = acaStats.deductible;

            if (selectedPlanTier === 'bronze') { 
                displayPlanName = "Bronze High-Deductible Tier"; 
                displayPrem = Math.round(acaStats.monthlyACA * 0.7); 
                displayDed = Math.round(acaStats.deductible * 2.2);
            } else if (selectedPlanTier === 'gold') { 
                displayPlanName = "Gold Preferred Copay Tier"; 
                displayPrem = Math.round(acaStats.monthlyACA * 1.5); 
                displayDed = Math.round(acaStats.deductible * 0.4); 
            }

            const activeCategoryEl = document.getElementById('active-fmo-category-title');
            if (activeCategoryEl) activeCategoryEl.textContent = `${displayPlanName} (${carrierObj.category})`;

            const annuityAlloc = targetAnnualContribution;

            const annuityAllocEl = document.getElementById('annuity-alloc-display');
            if (annuityAllocEl) annuityAllocEl.textContent = '$' + Math.round(annuityAlloc).toLocaleString() + ' / yr';
            const annuityMaxEl = document.getElementById('annuity-max-display');
            if (annuityMaxEl) annuityMaxEl.textContent = `Max Cap: $${currentIRSLimit.toLocaleString()} (${isMarried ? 'Married Joint' : 'Single'})`;

            const bannerEl = document.getElementById('recommended-plan-name-banner');
            if (bannerEl) bannerEl.textContent = `${displayCarrier}`;
            const premEl = document.getElementById('live-aca-premium');
            if (premEl) premEl.textContent = '$' + displayPrem + ' / mo';
            const dedEl = document.getElementById('live-aca-deductible');
            if (dedEl) dedEl.textContent = '$' + displayDed.toLocaleString() + ' / yr';
            const oopEl = document.getElementById('live-aca-oop');
            if (oopEl) oopEl.textContent = '$' + acaStats.oopMax.toLocaleString() + ' / yr';
            const avEl = document.getElementById('live-aca-av');
            if (avEl) avEl.textContent = acaStats.avText;
            const fplDisplay = document.getElementById('live-fpl-display');
            if (fplDisplay) fplDisplay.textContent = '~' + optimizedFPL + '% FPL';

            const marginalTaxRate = getMarginalTaxRate(currentAGI, clientData.filingStatus);
            const taxRelief = Math.round(targetAnnualContribution * marginalTaxRate);
            const deductibleRiskSaved = Math.max(0, unoptStats.deductible - acaStats.deductible);
            const premiumSubsidiesGained = Math.max(0, (unoptStats.monthlyACA - acaStats.monthlyACA) * 12);
            
            const totalComprehensiveSavings = targetAnnualContribution + taxRelief + deductibleRiskSaved + premiumSubsidiesGained;

            const contribTaxSavingsEl = document.getElementById('contrib-tax-savings-display');
            if (contribTaxSavingsEl) contribTaxSavingsEl.textContent = '+$' + taxRelief.toLocaleString() + ' / yr';
            const contribBracketEl = document.getElementById('contrib-tax-bracket-lbl');
            if (contribBracketEl) contribBracketEl.textContent = `Based on ${Math.round(marginalTaxRate * 100)}% Tax Bracket`;

            const origAgiTxt = document.getElementById('sim-orig-agi-txt');
            if (origAgiTxt) origAgiTxt.textContent = '$' + Math.round(currentAGI).toLocaleString();
            
            const origPlanTxt = document.getElementById('sim-orig-plan-txt');
            if (origPlanTxt) origPlanTxt.innerHTML = `${unoptStats.avText} (<span class="text-slate-300 font-bold">${displayCarrier}</span>)`;

            const adjAgiTxt = document.getElementById('sim-adj-agi-txt');
            if (adjAgiTxt) adjAgiTxt.textContent = '$' + Math.round(optimizedMAGI).toLocaleString();

            const adjPlanTxt = document.getElementById('sim-adj-plan-txt');
            if (adjPlanTxt) adjPlanTxt.innerHTML = `${acaStats.avText} (<span class="text-emerald-300 font-bold">${displayCarrier}</span>)`;

            const acaSavingsPill = document.getElementById('sim-aca-savings-pill');
            if (acaSavingsPill) acaSavingsPill.textContent = premiumSubsidiesGained > 0 ? `+$${premiumSubsidiesGained}/yr Subsidy Shift` : `$0 Subsidy Shift`;

            const simFplEl = document.getElementById('sim-fpl-pct-display');
            if (simFplEl) simFplEl.textContent = '~' + optimizedFPL + '% FPL';
            const simSavEl = document.getElementById('sim-total-savings-display');
            if (simSavEl) simSavEl.textContent = '+$' + Math.round(totalComprehensiveSavings).toLocaleString() + ' / yr';

            const sp1 = document.getElementById('sp1-retained');
            if (sp1) sp1.textContent = '+$' + targetAnnualContribution.toLocaleString() + ' / yr';
            const sp2 = document.getElementById('sp2-subsidies');
            if (sp2) sp2.textContent = '-$' + deductibleRiskSaved.toLocaleString() + ' Deductible Risk';
            const sp3 = document.getElementById('sp3-tax-relief');
            if (sp3) sp3.textContent = '+$' + taxRelief.toLocaleString() + ' / yr (' + Math.round(marginalTaxRate * 100) + '% Bracket)';

            const totalValueCaptured = targetAnnualContribution + taxRelief;
            const netSavings = totalValueCaptured + deductibleRiskSaved + premiumSubsidiesGained;
            const kpiVal = document.getElementById('kpi-value-captured');
            if (kpiVal) kpiVal.textContent = '+$' + totalValueCaptured.toLocaleString() + ' / yr';
            const kpiSwing = document.getElementById('kpi-net-swing');
            if (kpiSwing) kpiSwing.textContent = '+$' + netSavings.toLocaleString() + ' / yr';

            const monthlyDPC = Math.round(85 * (hhSize > 1 ? 1.5 : 1.0));
            const netMonthlyOverhead = monthlyDPC + displayPrem;
            const legacyMonthlyOverhead = Math.round(600 * famMult);

            const overheadEl = document.getElementById('kpi-overhead');
            if (overheadEl) overheadEl.textContent = '$' + netMonthlyOverhead + ' / mo';
            const overheadSub = document.getElementById('kpi-overhead-sub');
            if (overheadSub) overheadSub.textContent = '$' + monthlyDPC + ' DPC Care + $' + displayPrem + ' ACA Plan';

            const kpiMagi = document.getElementById('kpi-magi');
            if (kpiMagi) kpiMagi.textContent = '$' + optimizedMAGI.toLocaleString();
            const csrBadge = document.getElementById('kpi-csr-status-badge');
            if (csrBadge) csrBadge.textContent = acaStats.avText + ' Subsidized';

            const bPretax = document.getElementById('breakdown-pretax');
            if (bPretax) bPretax.textContent = '$' + targetAnnualContribution.toLocaleString();
            const bTax = document.getElementById('breakdown-tax');
            if (bTax) bTax.textContent = '$' + taxRelief.toLocaleString();
            const bDed = document.getElementById('breakdown-ded');
            if (bDed) bDed.textContent = '$' + deductibleRiskSaved.toLocaleString();
            const bSub = document.getElementById('breakdown-subsidy');
            if (bSub) bSub.textContent = '$' + premiumSubsidiesGained.toLocaleString();
            const bTotal = document.getElementById('breakdown-total-benefit');
            if (bTotal) bTotal.textContent = '+$' + Math.round(totalComprehensiveSavings).toLocaleString() + ' / yr';

            if (allocChart) {
                const legacyAnnualPrem = legacyMonthlyOverhead * 12;
                const atlasAnnualPrem = netMonthlyOverhead * 12;
                const legacyDeductible = unoptStats.deductible;
                const atlasDeductible = acaStats.deductible;
                const legacySubsidies = 0;
                const atlasSubsidies = premiumSubsidiesGained;
                const legacyOopMax = unoptStats.oopMax;
                const atlasOopMax = acaStats.oopMax;

                allocChart.data.datasets[0].data = [legacyAnnualPrem, legacyDeductible, legacySubsidies, legacyOopMax];
                allocChart.data.datasets[1].data = [atlasAnnualPrem, atlasDeductible, atlasSubsidies, atlasOopMax];
                allocChart.update();
            }
        }
